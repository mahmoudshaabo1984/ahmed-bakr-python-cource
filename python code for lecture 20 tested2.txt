# استيراد مكتبة random للأرقام العشوائية
import random
# طباعة معلومات عن كائن المكتبة
print(random)
# عرض جميع الدوال والخصائص المتاحة في مكتبة random
print(dir(random))


# استخدام دالة random() لتوليد رقم عشوائي عشري بين 0 و 1
import random
print(random.random())


# استخدام دالة randint() لتوليد أرقام صحيحة عشوائية
import random
# تصحيح: تغيير النطاق ليكون أكثر فائدة (كان 50, 50)
print(random.randint(1, 50))  # رقم عشوائي بين 1 و 50
print(random.randint(10, 600))  # رقم عشوائي بين 10 و 600


# استيراد دوال محددة من المكتبة بدلاً من استيراد المكتبة كاملة
# هذا يسمح باستخدام الدوال مباشرة دون كتابة اسم المكتبة
from random import random, randint

print(random())  # رقم عشوائي عشري بين 0 و 1
print(randint(10, 600))  # رقم صحيح عشوائي بين 10 و 600


# استيراد المكتبة باسم مختصر لتسهيل الاستخدام
import random as rm

print(rm.random())  # رقم عشوائي عشري بين 0 و 1
print(rm.randint(10, 600))  # رقم صحيح عشوائي بين 10 و 600

import random

def say_hello(name):
    """دالة لطباعة رسالة ترحيب بسيطة"""
    print(f"hello {name}")

def say_hi(name):
    """دالة لطباعة تحية رسمية"""
    print(f"hi mr {name}")

def show_random_num():
    """دالة لعرض أرقام عشوائية"""
    # تصحيح: إضافة طباعة رقمين كما يذكر التعليق
    print(f"رقم عشري عشوائي: {random.random()}")
    print(f"رقم صحيح عشوائي بين 10 و 600: {random.randint(10, 600)}")
